3. Sửa migrate.ts

Xóa toàn bộ phần:

TypeScript
import dotenv from 'dotenv';
import path from 'node:path';

dotenv.config(...);

Dùng file này:

TypeScript
import { resolve } from 'node:path';

import { drizzle } from 'drizzle-orm/mysql2';
import { migrate } from 'drizzle-orm/mysql2/migrator';
import mysql from 'mysql2/promise';

import { getDatabaseConfig } from './database.config';

async function runMigration(): Promise<void> {
  const config = getDatabaseConfig();

  const pool = mysql.createPool({
    host: config.host,
    port: config.port,
    user: config.user,
    password: config.password,
    database: config.database,
    connectionLimit: config.connectionLimit,
    timezone: config.timezone,
    waitForConnections: true,
    queueLimit: 0,
    charset: 'utf8mb4',
  });

  const database = drizzle(pool);

  try {
    console.log(
      `Connecting to MySQL ${config.host}:${config.port}/${config.database}`,
    );

    await migrate(database, {
      migrationsFolder: resolve(process.cwd(), 'drizzle'),
    });

    console.log('Database migration completed successfully.');
  } catch (error) {
    console.error('Database migration failed:', error);
    process.exitCode = 1;
  } finally {
    await pool.end();
  }
}

void runMigration();

Khi chạy qua filter, process.cwd() sẽ là:

nên thư mục migration đúng là:

4. Sửa seed.ts

Cũng xóa toàn bộ phần dotenv.config() trong file này. Biến môi trường đã được script nạp trước khi tsx chạy.

Phần khởi tạo database nên là:

TypeScript
import { drizzle } from 'drizzle-orm/mysql2';
import mysql from 'mysql2/promise';

import { getDatabaseConfig } from './database.config';
import * as schema from './schema';

async function seed(): Promise<void> {
  const config = getDatabaseConfig();

  const pool = mysql.createPool({
    host: config.host,
    port: config.port,
    user: config.user,
    password: config.password,
    database: config.database,
    connectionLimit: config.connectionLimit,
    timezone: config.timezone,
    waitForConnections: true,
    queueLimit: 0,
    charset: 'utf8mb4',
  });

  const database = drizzle(pool, {
    schema,
    mode: 'default',
  });

  try {
    console.log('Starting WorkLink seed data...');

    // Giữ nguyên các câu lệnh insert dữ liệu hiện có tại đây.

    console.log('Seed data completed successfully.');
  } catch (error) {
    console.error('Seed data failed:', error);
    process.exitCode = 1;
  } finally {
    await pool.end();
  }
}

void seed();
